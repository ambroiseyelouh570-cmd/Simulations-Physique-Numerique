clear; close; clc
Nx=2048; Nt=4000; dt=0.01/2; dx=0.05; m=1;
x = 1:Nx; x=((x-Nx/2)*dx)';
L=Nx*dx;
% barriere :
largeur=2;
q = round(largeur/dx/2); V = zeros(Nx,1); P = round(Nx/2);
V(P-q:P+q) = 1.5;
plot(x,V,'b')
% paquet d'onde
kk0 = 2;
xx0 = -20;
sigma = 3;
psi0 = (exp(-(x-xx0).*(x-xx0)/4/sigma/sigma+1j.*kk0.*x)/(2*pi*sigma^2)^.25);
% pour le temps : iteration sur i
dkx=2*pi/Nx/dx; k0=-pi/dx; kx=1:Nx; kx=(k0+kx*dkx)';
psix = zeros(Nx,Nt); psik = zeros(Nx,Nt);
psimk = zeros(Nx);
psix(:,1)=psi0;
psimx=dx/sqrt(2*pi).*psi0.*exp(-1j*k0.*x);
for i=1:Nt-1
  psimx=psimx.*exp(-1j.*dt/2.*V);
  psimk=fft(psimx);
  psimk=psimk.*exp(-1j.*kx.*kx*dt/2/m);
  psimx=ifft(psimk);
  psimx=psimx.*exp(-1j*dt/2.*V);
  psix(:,i+1)=sqrt(2*pi)/dx.*psimx.*exp(1j*k0.*x);
  psik(:,i+1)=psimk.*exp(-1j*x(1)*kx);
end
figure
for i=101:100:4000
  subplot(2,1,1), hold on
  plot(x,abs(psix(:,i-100)),'w')
  plot(x,real(psix(:,i-100)),'w')
  plot(x,abs(psix(:,i)),'r')
  plot(x,real(psix(:,i)),'r')
  plot(x,V,'b')
  axis([-50 50 -1 1])
  xlabel("x")
  ylabel("|psi|^2")
  grid()
  hold off
  subplot(2,1,2), hold on
  plot(kx,abs(psik(:,i-100)),'w')
%  plot(kx,real(psik(:,i-100)),'w')
  plot(kx,abs(psik(:,i)),'r')
%  plot(kx,real(psik(:,i)),'b')
  axis([-10 10 0 2])
  hold off
  pause(.1)
end
